
==================================================
/etc/nginx/nginx.conf
==================================================
worker_processes auto;

user root;

events {}

http {
	access_log off;
	log_format openwrt
		'$request_method $scheme://$host$request_uri => $status'
		' (${body_bytes_sent}B in ${request_time}s) <- $http_referer';

	include mime.types;
	default_type application/octet-stream;
	sendfile on;

	client_max_body_size 128M;
	large_client_header_buffers 2 1k;

	gzip on;
	gzip_vary on;
	gzip_proxied any;

	root /www;

	server { #see uci show 'nginx._lan'
		listen 443 ssl default_server;
		listen 80;
		server_name _lan;
		include restrict_locally;
		include /soft/config/nginx.locations;
		ssl_certificate /etc/nginx/conf.d/_lan.crt;
		ssl_certificate_key /etc/nginx/conf.d/_lan.key;
		ssl_session_cache shared:SSL:32k;
		ssl_session_timeout 64m;
	}

	server { #see uci show 'nginx._subvendo'
		listen 4455 ssl default_server;
		server_name _subvendo;
		include restrict_locally;
		include /soft/config/subvendo.locations;
		ssl_certificate /etc/nginx/conf.d/_lan.crt;
		ssl_certificate_key /etc/nginx/conf.d/_lan.key;
		ssl_session_cache shared:SSL:32k;
		ssl_session_timeout 64m;
	}

	server { #see uci show 'nginx.redirect_pppoe'
		listen 8081;
		server_name _pppoe;
		include /soft/config/pppoe.locations;
	}

	server { #see uci show 'nginx.media_server'
		listen 4400;
		server_name _media;
		include /soft/config/media.locations;
	}

	include conf.d/*.conf;
}

root /tmp/i/public;

location ~ \.(mp3|ico|map|css|jpg|png|js|woff|woff2|ttf) {
        limit_conn perip 10;
        limit_req zone=ip burst=30 delay=8;
        root /tmp/i/public;
}

location /admin {
    limit_conn perip 10;
    limit_req zone=ip burst=30 delay=8;
    try_files $uri @backend;
}

location /reseller {
    limit_conn perip 10;
    limit_req zone=ip burst=30 delay=8;
    try_files $uri @backend;
}

location / {
    limit_conn perip 10;
    limit_req zone=ip burst=30 delay=8;
    return 302 http://$host/admin;
}

location @backend {
    proxy_pass http://backend;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header Host $host;
	proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
}

location /tty-terminal {
    proxy_pass http://127.0.0.1:7681/$1;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header Host $host;
	proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
}
